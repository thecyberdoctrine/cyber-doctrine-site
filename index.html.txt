// Tab switcher in hero
function switchTab(btn, tab) {
  document.querySelectorAll('.cd-tab').forEach(t => t.classList.remove('active'));
  btn.classList.add('active');
  document.getElementById('tab-b2b').style.display = tab === 'b2b' ? 'block' : 'none';
  document.getElementById('tab-b2c').style.display = tab === 'b2c' ? 'block' : 'none';
}

// Pricing tier selector
function selectTier(el, url) {
  document.querySelectorAll('.cd-price-tier').forEach(t => t.classList.remove('selected'));
  el.classList.add('selected');
  document.getElementById('cd-checkout-btn').href = url;
}

function riskNext(step) {
  document.querySelectorAll('[data-step]').forEach(s => s.classList.remove('active'));
  var target = document.querySelector('[data-step="'+step+'"]');
  if (target) target.classList.add('active');
  for (var i = 1; i <= 5; i++) {
    var p = document.getElementById('rp'+i);
    if (p) p.className = 'cd-progress-step' + (i < step ? ' done' : i === step ? ' active' : '');
  }
}

function calcRisk() {
  var score = 0;
  ['r2','r3','r4','r5','r6','r7','r8','r9','r10'].forEach(function(name) {
    var el = document.querySelector('input[name="'+name+'"]:checked');
    if (el) score += parseInt(el.value);
  });
  var level, color, title, desc, recs;
  if (score <= 8) {
    level='LOW RISK'; color='#15803d'; title='Strong Foundation';
    desc='Your organization has solid security practices in place. Continue to build on this foundation.';
    recs=['Consider advanced phishing simulations to test your team', 'Document and formalize existing policies', 'Explore compliance framework alignment'];
  } else if (score <= 18) {
    level='MODERATE RISK'; color='#d97706'; title='Room to Improve';
    desc='You have some controls in place but meaningful gaps remain that attackers can exploit.';
    recs=['Implement MFA across all critical systems immediately', 'Launch a formal security awareness training program', 'Run your first phishing simulation this quarter'];
  } else if (score <= 28) {
    level='HIGH RISK'; color='#dc2626'; title='Action Required';
    desc='Significant gaps exist. Your employees and data are at considerable risk right now.';
    recs=['Employee security awareness training is your #1 priority', 'Enforce MFA within 30 days', 'Develop an incident response plan', 'Consider the Cyber Doctrine 30-Day Pilot'];
  } else {
    level='CRITICAL RISK'; color='#7c3aed'; title='Immediate Action Needed';
    desc='Your organization is highly vulnerable. A breach is a matter of when, not if, without immediate action.';
    recs=['Start the Cyber Doctrine Pilot Program this week', 'Emergency: Enable MFA on all accounts today', 'Implement email filtering immediately', 'Develop an incident response plan'];
  }
  var pct = Math.round((1 - score/36) * 100);
  document.getElementById('risk-steps').style.display = 'none';
  document.getElementById('risk-result').innerHTML = `
    <div class="cd-result-box">
      <h3>Your Risk Score</h3>
      <div class="cd-result-score" style="color:${color};">${pct}<span style="font-size:28px;">/100</span></div>
      <div style="display:inline-block;background:${color}20;color:${color};padding:4px 14px;border-radius:100px;font-size:12px;font-weight:700;letter-spacing:0.08em;margin-bottom:12px;">${level}</div>
      <div class="cd-result-meter"><div class="cd-result-meter-fill" style="width:${pct}%;"></div></div>
      <p><strong>${title}:</strong> ${desc}</p>
      <p style="font-weight:600;margin-top:14px;margin-bottom:8px;">Top Recommendations:</p>
      <ul class="cd-result-list">${recs.map(r=>`<li><span class="cd-stat-dot" style="background:${color};flex-shrink:0;"></span>${r}</li>`).join('')}</ul>
    </div>
    ${funnelBox()}`;
  document.getElementById('risk-result').style.display = 'block';
}

function calcBenchmark() {
  var ind = document.getElementById('b-industry').value;
  var size = document.getElementById('b-size').value;
  if (!ind || !size) { alert('Please complete all fields.'); return; }
  var total = 0;
  for (var i = 0; i < 8; i++) total += parseInt(document.getElementById('bq'+i).value);
  var avg = (total / 8).toFixed(1);
  var pct = Math.round((total / 40) * 100);
  var industryAvg = { Healthcare: 58, Finance: 65, Technology: 61, Retail: 49, Education: 45, Professional: 54 };
  var peer = industryAvg[ind] || 52;
  var diff = pct - peer;
  var diffText = diff >= 0 ? `<span style="color:#15803d;">+${diff}% above average</span>` : `<span style="color:#dc2626;">${diff}% below average</span>`;
  document.getElementById('bench-result').innerHTML = `
    <div class="cd-result-box" style="margin-top:24px;">
      <h3>Your Benchmark Score</h3>
      <div class="cd-result-score">${pct}<span style="font-size:28px;">/100</span></div>
      <div class="cd-result-meter"><div class="cd-result-meter-fill" style="width:${pct}%;"></div></div>
      <p><strong>Your Score:</strong> ${pct}% &nbsp;|&nbsp; <strong>${ind} Industry Average:</strong> ${peer}% &nbsp;|&nbsp; ${diffText}</p>
      <p style="margin-top:12px;">Your average practice rating is <strong>${avg}/5</strong>. ${pct < peer ? 'You are below your industry peers — targeted training and controls could close this gap quickly.' : 'You are performing above your peers — focus on maintaining and formalizing your practices.'}</p>
    </div>
    ${funnelBox()}`;
}

  var gaps = data.items.length - checked;
  document.getElementById('comp-result').innerHTML = `
    <div class="cd-result-box" style="margin-top:20px;">
      <h3>${data.name} Gap Analysis</h3>
      <div class="cd-result-score">${pct}<span style="font-size:28px;">%</span></div>
      <div class="cd-result-meter"><div class="cd-result-meter-fill" style="width:${pct}%;"></div></div>
      <p><strong>${checked}/${data.items.length} controls</strong> in place. You have <strong>${gaps} gap${gaps!==1?'s':''}</strong> to address.</p>
      <p style="margin-top:10px;">${pct >= 80 ? 'Strong compliance posture. Focus on the remaining gaps to achieve full readiness.' : pct >= 50 ? 'Moderate compliance. Significant gaps remain that could expose you to regulatory risk.' : 'Significant compliance gaps identified. Immediate action is recommended before an audit or breach.'}</p>
    </div>
    ${funnelBox()}`;
}

function calcROI() {
  var emp = parseInt(document.getElementById('roi-emp').value) || 25;
  var sal = parseInt(document.getElementById('roi-sal').value) || 65000;
  var ind = parseFloat(document.getElementById('roi-ind').value);
  var clickRate = parseInt(document.getElementById('roi-click').value) / 100;
  var avgBreachCost = 4450000 * ind;
  var breachProb = clickRate * 0.3;
  var annualRisk = avgBreachCost * breachProb;
  var downtimeHours = emp * 8 * 2;
  var downtimeCost = (sal / 2080) * downtimeHours;
  var totalExposure = annualRisk + downtimeCost;
  var trainingCost = emp <= 15 ? 1500 : emp <= 50 ? 3500 : 5000;
  var reducedRisk = totalExposure * 0.70;
  var roi = Math.round(((reducedRisk - trainingCost) / trainingCost) * 100);
  var fmt = n => '$' + Math.round(n).toLocaleString();
  document.getElementById('roi-result').innerHTML = `
    <div class="cd-result-box" style="margin-top:24px;">
      <h3>Your ROI Analysis</h3>
      <div style="display:grid;grid-template-columns:1fr 1fr;gap:14px;margin:16px 0;">
        <div style="background:white;border-radius:10px;padding:16px;text-align:center;">
          <div style="font-size:11px;text-transform:uppercase;letter-spacing:0.1em;color:var(--text-muted);margin-bottom:4px;">Annual Risk Exposure</div>
          <div style="font-family:var(--font-display);font-size:28px;color:#dc2626;">${fmt(totalExposure)}</div>
        </div>
        <div style="background:white;border-radius:10px;padding:16px;text-align:center;">
          <div style="font-size:11px;text-transform:uppercase;letter-spacing:0.1em;color:var(--text-muted);margin-bottom:4px;">Risk Reduction (70%)</div>
          <div style="font-family:var(--font-display);font-size:28px;color:#15803d;">${fmt(reducedRisk)}</div>
        </div>
        <div style="background:white;border-radius:10px;padding:16px;text-align:center;">
          <div style="font-size:11px;text-transform:uppercase;letter-spacing:0.1em;color:var(--text-muted);margin-bottom:4px;">Training Investment</div>
          <div style="font-family:var(--font-display);font-size:28px;color:var(--blue);">${fmt(trainingCost)}</div>
        </div>
        <div style="background:white;border-radius:10px;padding:16px;text-align:center;">
          <div style="font-size:11px;text-transform:uppercase;letter-spacing:0.1em;color:var(--text-muted);margin-bottom:4px;">Projected ROI</div>
          <div style="font-family:var(--font-display);font-size:28px;color:#15803d;">${roi}%</div>
        </div>
      </div>
      <p style="font-size:13px;">Based on ${emp} employees, a ${Math.round(clickRate*100)}% phishing click rate, and industry breach cost data. A 70% risk reduction is the average result of a structured training program.</p>
    </div>
    ${funnelBox()}`;
}

    </div>`;

  var terminal = document.getElementById('sim-terminal');
  var i = 0;
  function typeNext() {
    if (i < scene.steps.length) {
      var step = scene.steps[i];
      terminal.innerHTML += `<span class="${step.t || ''}">${step.msg}</span>\n`;
      terminal.scrollTop = terminal.scrollHeight;
      i++;
      setTimeout(typeNext, 900 + Math.random() * 400);
    } else {
      document.getElementById('sim-outcome').style.display = 'block';
    }
  }
  setTimeout(typeNext, 500);
}
